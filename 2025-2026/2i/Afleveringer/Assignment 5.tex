
\begin{opgavetekst}{Problem 1}
\end{opgavetekst}
\begin{delopgave}{1}
	Determine the first 5 terms of the following sequences
	\begin{align*}
		&1) \ \left\{(-2)^n\right\}_{n=1}^\infty & &2) \ a_n = 
		\begin{cases}
			-2 \ &\textnormal{ if }n = 1, \\
			2a_{n-1} + 4 \ &\textnormal{ if }n > 1.
		\end{cases}
		 \\
		&3) \ b_n = 9n - 7 & &4) \ \left\{ \frac{2^n}{4^{n-1}} \right\}_{n=1}^\infty  \\
	\end{align*}
\end{delopgave}

\begin{opgavetekst}{Problem 4}
\end{opgavetekst}
\begin{delopgave}{1}
	Determine an explicit formula for the following sequences. 
	\begin{align*}
		&1) \ -4, 3, 10, 17, 24, \hdots  & &2) \ 300, 150, 75, 37.5, \hdots \\
		&3) \ 
		a_n = 
		\begin{cases}
			-\frac{1}{3} \ &\textnormal{ if } n = 1, \\
			-\frac{2}{3}a_{n-1} \ &\textnormal{ if } n>1.				
		\end{cases}
		& &4) \ 
		b_n = 
		\begin{cases}
			2 \ &\textnormal{ if } n = 1, \\
			(b_{n-1})^2 \ &\textnormal{ if } n > 1. \\
		\end{cases}
	\end{align*}
\end{delopgave}

\begin{opgavetekst}{Problem 5}
	The number of bacteria in some container is at time $t = 1$ equal to $7.6$ million. Every hour the number of 
	bacteria grows by $15\%$.
	The sequence $b_t$ gives the number of bacteria in the container after $t$ hours. 
\end{opgavetekst}
\begin{delopgave}{1}
	Determine an explicit formula for the sequence $b_t$.
\end{delopgave}
\begin{delopgave}{2}
	Determine a recursive formula for the sequence $b_t$.
\end{delopgave}
