\begin{opgavetekst}{Problem 1}
\end{opgavetekst}
\begin{delopgave}{1}
	Determine the derivative of the following functions.
	\begin{align*}
		&1) \ f(x) = 2x^5    &  &2) \ f(x) = 2.5x^{\frac{1}{5}}-9x^{-1}   \\
		&3) \ f(x) = 10\sqrt{x} + \frac{1}{7}x^{\frac{5}{6}}    &  &4) \  f(x) = \frac{9}{x^6}  \\
	\end{align*}
\end{delopgave}


\begin{opgavetekst}{Problem 2}
\end{opgavetekst}
\begin{delopgave}{1}
	Determine the slope of the tangent of the following functions at the given points.
	\begin{align*}
		&1) \ f(x) = x^2 + 4x -10, & &P(-5,f(-5)) \\
		&2) \ f(x) = \sin(x), & &P(\pi,f(\pi)) \\
		&3) \ f(x) = 32\sqrt{x}, & &P(16,f(16)) \\
		&4) \ f(x) = x^{10}-5x^8, & &P(-1,f(-1)) \\
	\end{align*}
\end{delopgave}

\begin{opgavetekst}{Problem 3}
	A function $f$ is given by
	\begin{align*}
		f(x) = x^3 - 3x^2 + 7.
	\end{align*}
\end{opgavetekst}
\begin{delopgave}{1}
	Determine an equation for the tangent line through the point $(2,f(2))$ on the form $y = mx + b$. 
\end{delopgave}


\begin{opgavetekst}{Problem 4}
\end{opgavetekst}
\begin{delopgave}{1}
	Determine any stationary points for each of the following functions. Denote whether the points are maxima, minima or
	points of inflection.
	\begin{align*}
		&1) \ f(x) = x^4 - 32x^2 + 10 \\
		&2) \ f(x) = \frac{1}{3}x^3 + \frac{1}{2}x^2 - 6x + 4\\
		&3) \ f(x) = \frac{2}{5}x^5 - 27x^2 \\
	\end{align*}
\end{delopgave}

\begin{opgavetekst}{Problem 5}
A function $g$ is given by
\begin{align*}
	g(x) = 5x^2 - 3x.
\end{align*}
\end{opgavetekst}
\begin{delopgave}{1}
	Show, by applying the definition of the differential quotient, that 
	\begin{align*}
		g'(2) = 17.
	\end{align*}
	(Hint: Find the slope of the secant line in $x_0 = 2$ and let $h \to 0$.)
\end{delopgave}	

\begin{opgavetekst}{Problem 6}
	A second-degree polynomial $p$ is in general given by
	\begin{align*}
		p(x) = ax^2 + bx + c
	\end{align*}
\end{opgavetekst}
\begin{delopgave}{1}
	Show that the vertex of $p$ is given by $\left(\frac{-b}{2a},\frac{-d}{4a} \right)$, where $d$ is the discriminant 
	of $p$. 
\end{delopgave}
