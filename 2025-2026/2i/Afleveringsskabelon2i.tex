\begin{opgavetekst}{Problem 1}
	The function $f$ is given by
	\begin{align*}
		f(x) = 5x^2 -10x + 9.
	\end{align*}
\end{opgavetekst}
\begin{delopgave}{1}
	Determine an antiderivative of $f$ and show that your function is an antiderivative.
\end{delopgave}

\begin{opgavetekst}{Problem 2}
	The function $g$ is given by
	\begin{align*}
		g(x) = \frac{7}{x} + \frac{3}{5}, \ x> 0.
	\end{align*}
\end{opgavetekst}
\begin{delopgave}{1}
	Determine $\int g(x) dx$.
\end{delopgave}

\begin{opgavetekst}{Problem 3}
	The indeterminate integral of the function $\ln(x)$ is given by
	\begin{align*}
		\int \ln(x) dx = x\ln(x) - x + C
	\end{align*}
\end{opgavetekst}
\begin{delopgave}{1}
	Give a proof of this rule. 
\end{delopgave}


\begin{opgavetekst}{Problem 4}
	The function $h$ is given by
	\begin{align*}
		h(t) = \frac{t^3}{\sqrt{t^4+1}}
	\end{align*}
\end{opgavetekst}
\begin{delopgave}{1}
	Find $\int h(t) dt$.
\end{delopgave}

\begin{opgavetekst}{Problem 5}	
\end{opgavetekst}

	\begin{align*}
		&1) \int_1^3 3x^2+2x-1 dx & &2) \int_{9}^{16} \frac{1}{2\sqrt{x}} dx
	\end{align*}
\end{delopgave}

\begin{opgavetekst}{Problem 6}
	A function $\varphi$ is given by
	\begin{align*}
		\varphi(x) = \frac{e^{4\sin(x)}\cos(x)}{2}
	\end{align*}
\end{opgavetekst}
\begin{delopgave}{1}
	Determine 
	\begin{align*}
		\int_0^\pi \varphi(x) dx
	\end{align*}
\end{delopgave}

\begin{opgavetekst}{Problem 7}
	The functions $f$ and $g$ given by 
	\begin{align*}
		f(x) = -x^2 + 2x +3
	\end{align*}
	and 
	\begin{align*}
		g(x) = 3x-9
	\end{align*}
	bound a region $M$.
\end{opgavetekst}
\begin{delopgave}{1}
	Determine the intersections between $f$ and $g$
\end{delopgave}
\begin{delopgave}{2}
	Determine the roots of $f$
\end{delopgave}
\begin{delopgave}{3}
	Determine the area of $M$. 
\end{delopgave}
