\subsection*{Opgave 1}

En differentialligning er givet ved
\begin{align*}
	y' = 5xy + y^2.
\end{align*}
En funktion $f$ er løsning til differentialligningen.
\begin{enumerate}[label=\roman*)]
	\item Bestem linjeelementet i punktet $P(3,-2)$.
	\item Bestem ligningen for tangenten til $f$ i punktet $P(3,-2)$.
\end{enumerate}

\subsection*{Opgave 2}
En differentialligning har følgende hældningsfelt. 
\begin{center}
	\begin{tikzpicture}
	\begin{axis}[
		xmin = -5, xmax = 5,
		ymin = -5, ymax = 5,
		axis lines = center,
		xlabel = $x$,
		ylabel = $y$,
	]
		\foreach \xVal in {-5,-4.5,...,5}{
			\foreach \yVal in {-5,-4.5,...,5}{
				\pgfmathsetmacro{\sizescale}{0.15};
				\pgfmathsetmacro{\slope}{0.35*\yVal*(5 - \yVal)};
				\pgfmathsetmacro{\angle}{atan(\slope)};
				\pgfmathsetmacro{\xlen}{\sizescale*cos(\angle)};
				\pgfmathsetmacro{\ylen}{\sizescale*sin(\angle)};
				\edef\temp{\noexpand\draw [color = SeaGreen] 
				(axis cs: \xVal-\xlen, \yVal-\ylen) -- (axis cs: \xVal + \xlen, \yVal + \ylen);}
				\temp
			}
		}
	\end{axis}
	\end{tikzpicture}
\end{center}
\begin{enumerate}[label=\roman*)]
	\item Tegn løsningskurven for løsningen $f$, der opfylder, at $f(2) = 2$. 
\end{enumerate}

\subsection*{Opgave 3}
En differentialligning er givet ved
\begin{align*}
	\frac{dy}{dx} = \frac{2y}{x} + y
\end{align*}
\begin{enumerate}[label = \roman*)]
	\item Vis, at funktionen $f$ givet ved
	\begin{align*}
		f(x) = x^2e^x 
	\end{align*}
	er en løsning til differentialligningen. 
\end{enumerate}

\section*{Opgaver med hjælpemidler}

\subsection*{Opgave 12}
En differentialligning er givet ved
\begin{align*}
	y' = \left( \cos(x)\cdot y\right)^2, \quad x > pi
\end{align*}
Funktionen $f$ er en partikulær løsning, der går gennem punktet $(4,3)$.
\begin{enumerate}[label=\roman*)]
	\item Bestem væksthastigheden i punktet $(4,3)$.
	\item Bestem en forskrift for løsningen $f$. 
	\item Find $f(10)$.
\end{enumerate}

\subsection*{Opgave 13}
I en by kan antallet af beboere $y$ i tusinde beskrives ved en differentialligning af typen 
\begin{align*}
	y' = ay(122-y).
\end{align*}
Tidsenheden er i år.
\begin{enumerate}[label=\roman*)]
	\item Bestem en partikulær løsning, når det oplyses, at $y(2) = 43$ og $y(13) = 71$. 
	\item Bestem antallet af beboere efter 30 år.
	\item Bestem det tidspunkt, hvor populationen i byen vokser hurtigst. 
\end{enumerate}
